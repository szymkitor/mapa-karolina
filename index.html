const viewer = document.getElementById('viewer'); const content = document.getElementById('content'); const img = document.getElementById('map-img'); let scale = 1; let pointX = 0; let pointY = 0; let isDragging = false; let startX = 0; let startY = 0; let initialPinchDistance = null; function setTransform() { content.style.transform = `translate(${pointX}px, ${pointY}px) scale(${scale})`; } function initMap() { if (!img.naturalWidth || !img.naturalHeight) return; const scaleX = viewer.clientWidth / img.naturalWidth; const scaleY = viewer.clientHeight / img.naturalHeight; scale = Math.min(scaleX, scaleY) * 0.95; pointX = (viewer.clientWidth - img.naturalWidth * scale) / 2; pointY = (viewer.clientHeight - img.naturalHeight * scale) / 2; setTransform(); } // Reaguj na załadowanie (oraz jeśli obrazek jest już w pamięci podręcznej) if (img.complete) { initMap(); } else { img.onload = initMap; } img.onerror = () => { console.error("BŁĄD: Nie znaleziono pliku SVG pod wskazaną ścieżką!"); }; img.addEventListener('dragstart', (e) => e.preventDefault()); // --- OBSŁUGA MYSZY (DESKTOP) --- viewer.addEventListener('mousedown', (e) => { if (e.button !== 0) return; isDragging = true; startX = e.clientX - pointX; startY = e.clientY - pointY; }); window.addEventListener('mousemove', (e) => { if (!isDragging) return; e.preventDefault(); pointX = e.clientX - startX; pointY = e.clientY - startY; setTransform(); }); window.addEventListener('mouseup', () => { isDragging = false; }); viewer.addEventListener('wheel', (e) => { e.preventDefault(); const xs = (e.clientX - pointX) / scale; const ys = (e.clientY - pointY) / scale; const delta = -e.deltaY; if (delta > 0) scale *= 1.15; else scale /= 1.15; scale = Math.max(0.05, Math.min(scale, 50)); pointX = e.clientX - xs * scale; pointY = e.clientY - ys * scale; setTransform(); }, { passive: false }); // --- OBSŁUGA DOTYKU (MOBILNE) --- viewer.addEventListener('touchstart', (e) => { if (e.touches.length === 1) { isDragging = true; startX = e.touches[0].clientX - pointX; startY = e.touches[0].clientY - pointY; } else if (e.touches.length === 2) { isDragging = false; initialPinchDistance = Math.hypot( e.touches[0].clientX - e.touches[1].clientX, e.touches[0].clientY - e.touches[1].clientY ); } }, { passive: true }); viewer.addEventListener('touchmove', (e) => { if (isDragging && e.touches.length === 1) { pointX = e.touches[0].clientX - startX; pointY = e.touches[0].clientY - startY; setTransform(); } else if (e.touches.length === 2 && initialPinchDistance) { e.preventDefault(); const currentDistance = Math.hypot( e.touches[0].clientX - e.touches[1].clientX, e.touches[0].clientY - e.touches[1].clientY ); const centerTouchX = (e.touches[0].clientX + e.touches[1].clientX) / 2; const centerTouchY = (e.touches[0].clientY + e.touches[1].clientY) / 2; const xs = (centerTouchX - pointX) / scale; const ys = (centerTouchY - pointY) / scale; const zoomFactor = currentDistance / initialPinchDistance; scale *= zoomFactor; scale = Math.max(0.05, Math.min(scale, 50)); pointX = centerTouchX - xs * scale; pointY = centerTouchY - ys * scale; initialPinchDistance = currentDistance; setTransform(); } }, { passive: false }); viewer.addEventListener('touchend', (e) => { if (e.touches.length < 2) initialPinchDistance = null; if (e.touches.length === 0) isDragging = false; });
